cmake_minimum_required(VERSION 3.20)
project(SharedLoot VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(SharedLoot SHARED src/shared_loot.cpp)
target_include_directories(SharedLoot PUBLIC include)
target_compile_definitions(SharedLoot PRIVATE SL_BUILD)
add_library(session_router STATIC src/session_router.cpp src/inventory_probe.cpp src/shop_purchase.cpp)
target_include_directories(session_router PUBLIC include)
add_library(rowa_transport STATIC src/rowa_transport.cpp)
target_include_directories(rowa_transport PUBLIC include)
if(WIN32)
  set_target_properties(SharedLoot PROPERTIES PREFIX "")
  target_link_libraries(rowa_transport PUBLIC ws2_32)
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The observer supports Windows x64 only")
  endif()
  add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
  target_include_directories(minhook PUBLIC third_party/minhook/include)
  target_sources(SharedLoot PRIVATE src/diagnostics_win.cpp src/steam_bridge_win.cpp)
  target_link_libraries(SharedLoot PRIVATE minhook bcrypt rowa_transport session_router)
endif()
if(MSVC)
  target_compile_options(SharedLoot PRIVATE /W4 /WX)
else()
  target_compile_options(SharedLoot PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE SharedLoot)
  add_test(NAME core COMMAND core_tests)
  add_executable(packet_tests tests/packet_tests.cpp)
  target_include_directories(packet_tests PRIVATE include)
  add_test(NAME packet COMMAND packet_tests)
  add_executable(source_filter_tests tests/source_filter_tests.cpp)
  target_include_directories(source_filter_tests PRIVATE include)
  target_link_libraries(source_filter_tests PRIVATE session_router)
  add_test(NAME source_filter COMMAND source_filter_tests)
  add_executable(session_router_tests tests/session_router_tests.cpp)
  target_link_libraries(session_router_tests PRIVATE session_router)
  add_test(NAME session_router COMMAND session_router_tests)
  add_executable(inventory_probe_tests tests/inventory_probe_tests.cpp)
  target_link_libraries(inventory_probe_tests PRIVATE session_router)
  add_test(NAME inventory_probe COMMAND inventory_probe_tests)
  add_executable(delivery_buffer_tests tests/delivery_buffer_tests.cpp)
  target_link_libraries(delivery_buffer_tests PRIVATE session_router)
  add_test(NAME delivery_buffer COMMAND delivery_buffer_tests)
  add_executable(shop_purchase_tests tests/shop_purchase_tests.cpp)
  target_link_libraries(shop_purchase_tests PRIVATE session_router)
  add_test(NAME shop_purchase COMMAND shop_purchase_tests)
  add_executable(rowa_transport_tests tests/rowa_transport_tests.cpp)
  target_link_libraries(rowa_transport_tests PRIVATE rowa_transport)
  add_test(NAME native_transport COMMAND rowa_transport_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME diagnostic_network COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_network.py -v)
    set_tests_properties(diagnostic_network PROPERTIES ENVIRONMENT "SL_TEST_LIBRARY=$<TARGET_FILE:SharedLoot>" TIMEOUT 45)
    add_test(NAME native_peer_processes COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/native_peer_processes.py $<TARGET_FILE:rowa_transport_tests>)
    add_test(NAME native_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/native_protocol.py $<TARGET_FILE:rowa_transport_tests>)
    if(WIN32)
      add_test(NAME settings_file COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/settings_win.py $<TARGET_FILE:SharedLoot>)
      add_test(NAME diagnostic_hook COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/windows_diagnostics.py $<TARGET_FILE:SharedLoot>)
      add_library(mock_steam SHARED tests/fake_steam_api.cpp)
      target_link_libraries(mock_steam PRIVATE session_router)
      set_target_properties(mock_steam PROPERTIES PREFIX "" OUTPUT_NAME "steam_api64" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mock-steam")
      add_test(NAME steam_adapter_mock COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/steam_adapter_mock.py $<TARGET_FILE:mock_steam> $<TARGET_FILE:SharedLoot>)
    endif()
  endif()
endif()
